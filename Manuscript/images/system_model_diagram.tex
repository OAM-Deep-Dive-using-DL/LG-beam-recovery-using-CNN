% Standalone System Model Diagram (artistic vertical layout)
\documentclass[tikz,border=7pt]{standalone}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta,positioning,fit,shapes.geometric,backgrounds}

\begin{document}
\begin{tikzpicture}[
    font=\sffamily\small,
    >=Latex,
    node distance=5.5mm and 8mm,
    core/.style={
      draw, very thick, rounded corners=3pt, align=center,
      minimum width=48mm, minimum height=8mm, fill=blue!10
    },
    split/.style={
      diamond, draw, very thick, aspect=1.8, align=center,
      inner sep=1.2pt, fill=orange!20
    },
    bproc/.style={
      draw, thick, rounded corners=3pt, align=center,
      minimum width=42mm, minimum height=8mm, fill=green!10
    },
    result/.style={
      draw, thick, rounded corners=3pt, align=left,
      minimum width=56mm, minimum height=9mm, fill=gray!12
    },
    flow/.style={-Latex, very thick, color=black!80},
    cflow/.style={-Latex, very thick, color=teal!70!black},
    nflow/.style={-Latex, very thick, color=purple!70!black}
]

% Shared vertical pipeline
\node[core] (bits) {Information bits $\mathbf{u}$};
\node[core, below=of bits] (tx) {Coding + QPSK mapping};
\node[core, below=of tx] (oam) {OAM mode multiplexing};
\node[core, below=of oam] (chan) {Atmospheric channel: SSFM + phase screens + $C_n^2$ sweep};
\node[core, below=of chan] (rx) {Received field $E_{rx}$};
\node[split, below=7mm of rx] (obs) {Shared\\observation};

% Branches (left: classical, right: neural)
\node[bproc, below left=10mm and 28mm of obs] (coh) {Coherent modal observation};
\node[bproc, below=of coh] (mmse) {Pilot-aided LS ($\hat{\mathbf{H}}$) + MMSE\\+ phase correction + LDPC};
\node[result, below=of mmse] (cout) {\textbf{Classical outputs}\\Pre-/post-LDPC BER\\Condition number $\kappa(\hat{\mathbf{H}})$};

\node[bproc, below right=10mm and 28mm of obs] (img) {Intensity observation\\($128\times128\times1$)};
\node[bproc, below=of img] (fth) {Learned symbol mapping\\$f_{\theta}(\mathbf{I}) \rightarrow \hat{\mathbf{s}}$};
\node[result, below=of fth] (nout) {\textbf{Neural outputs}\\BER/SER by regime\\Constellation evolution};

% Flows
\draw[flow] (bits) -- (tx);
\draw[flow] (tx) -- (oam);
\draw[flow] (oam) -- (chan);
\draw[flow] (chan) -- (rx);
\draw[flow] (rx) -- (obs);

\draw[cflow] (obs) -- (coh);
\draw[cflow] (coh) -- (mmse);
\draw[cflow] (mmse) -- (cout);

\draw[nflow] (obs) -- (img);
\draw[nflow] (img) -- (fth);
\draw[nflow] (fth) -- (nout);

% Track containers + labels
\begin{scope}[on background layer]
  \node[
    draw=blue!50!black, rounded corners=4pt, very thick,
    fit=(bits)(obs), inner xsep=10pt, inner ysep=6pt,
    fill=blue!3
  ] {};
  \node[
    draw=teal!65!black, rounded corners=4pt, thick,
    fit=(coh)(mmse)(cout), inner sep=6pt, fill=teal!5,
    label={[text=teal!70!black]above:\textbf{Classical receiver track}}
  ] {};
  \node[
    draw=purple!65!black, rounded corners=4pt, thick,
    fit=(img)(fth)(nout), inner sep=6pt, fill=purple!5,
    label={[text=purple!70!black]above:\textbf{Neural receiver track}}
  ] {};
\end{scope}

\end{tikzpicture}
\end{document}
